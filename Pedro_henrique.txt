FROM pedidos p 
INNER JOIN clientes c ON p.id_cliente = c.id 
WHERE p.data_pedido >= '2026-01-01' AND p.data_pedido < '2027-01-01';

SELECT id, valor, status 
FROM transacoes

9°

SELECT DISTINCT c.id, c.nome, c.email 
FROM clientes c
JOIN pedidos p ON c.id = p.id_cliente
JOIN produtos pr ON p.id_produto = pr.id
WHERE pr.categoria = 'Eletronicos';

10°

SELECT c.id, c.nome, p.id AS pedido_id, p.valor_total 
FROM clientes c 
INNER JOIN pedidos p ON c.id = p.id_cliente 
WHERE p.status = 'faturado' AND p.valor_total > 500;
