-- QUERY 1: O "Caminhão de Lixo" (SELECT * sem restrição e sem limite) 
SELECT * FROM clientes ORDER BY data_cadastro DESC;

Query otimizada:
SELECT id_cliente, nome, data_cadastro FROM clientes ORDER BY data_cadastri DESC LIMIT 100;

-- QUERY 2: Violação da hierarquia de índice (Leftmost Prefix ignorado) -- Cenário: Tabela possui o índice composto (estado, cidade) 
SELECT id_cliente, rua, bairro FROM enderecos WHERE cidade = 'Campinas';

Query otimizada:
SELECT id_cliente, rua, bairro FROM enderecos WHERE estado = 'São Paulo' AND cidade = 'Campinas';

-- QUERY 3: Junção Tardia com Nested Loop pesado (Falta de filtragem prévia) 
SELECT p.id, c.nome, p.valor_total FROM pedidos p INNER JOIN clientes c ON p.id_cliente = c.id WHERE YEAR(p.data_pedido) = 2026;

Queru otimizada:
SELECT p.id, c.nome, p.valor_total FROM ( SELECT id, id_cliente, data_pedido, valor_total FROM pedidos WHERE data_pedido >= '2026-01-01' AND data_pedido <= '2026-12-31') AS p JOIN clientes c ON p.id_cliente = c.id;

-- QUERY 4: Desperdício de Covering Index (Leitura física da tabela sem necessidade) -- Cenário: Tabela possui o índice composto (estado, cidade)
SELECT * FROM userinfo WHERE estado = 'SP';

Query otimizada:
SELECT estado, cidade FROM userinfo WHERE estado = 'SP';


-- QUERY 5: A Query Monstro da Capa (Subconsulta correlacionada e DISTINCT desnecessário) 
SELECT DISTINCT c.id, c.nome, SUM(p.valor) AS total_gasto, (SELECT MAX(sp.data_pedido) FROM pedidos sp WHERE sp.id_cliente = c.id) AS data_ultimo_pedido FROM clientes c JOIN pedidos p ON c.id = p.id_cliente WHERE c.data_cadastro >= '2023-01-01' AND c.status = 'ativo' GROUP BY c.id, c.nome ORDER BY total_gasto DESC;

Query otimizada:
SELECT c.id, c.nome, SUM(p.valor) AS total_gasto, MAX(p.data_pedido) AS data_ultimo_pedido FROM clientes c JOIN pedidos p ON c.id = p.id_cliente WHERE c.data_cadastro >= '2023-01-01' AND c.status = 'ativo' GROUP BY c.id, c.nome ORDER BY total_gasto DESC;

-- QUERY 6: Produto Cartesiano Velado (CROSS JOIN por falta de condição de junção) 
SELECT p.nome, c.nome_categoria FROM produtos p, categorias c;

Query otimizada:
SELECT p.nome, c.nome_categoria FROM produtos p INNER JOIN categorias c ON p.id_categoria = c.id

-- QUERY 7: Destruição do uso de índice em Árvore B+ por aplicação de funções no WHERE -- Cenário: Coluna data_criacao possui índice 
SELECT id, status, total FROM faturas WHERE MONTH(data_criacao) = 3 AND YEAR(data_criacao) = 2026;

Query otimizada:
SELECT id, status, total FROM faturas WHERE data_criacao >= '2026-03-01' AND data_criacao <= '2026-03-31'

-- QUERY 8: Type Mismatch forçando conversão implícita e varredura completa -- Cenário: Coluna codigo_transacao é VARCHAR(36) indexada
SELECT id, valor, status FROM transacoes WHERE codigo_transacao = 9845720194;

Query otimizada:
SELECT id, valor, status FROM transacoes WHERE codigo_transacao = '9845720194';

-- QUERY 9: Gargalo de I/O por Subconsultas aninhadas com operador IN 
SELECT id, nome, email FROM clientes WHERE id IN ( SELECT id_cliente FROM pedidos WHERE id_produto IN ( SELECT id FROM produtos WHERE categoria = 'Eletronicos' ) );

Query otimizada:
SELECT DISTINCT c.id, c.nome, c.email FROM cliente c JOIN pedidos p ON c.id = p.id_cliente JOIN produtos pd ON p.id_categoria = pd.id WHERE pd.categoria = 'Eletronicos';

-- QUERY 10: LEFT JOIN desnecessário limitando o otimizador no Nested Loop 
SELECT c.id, c.nome, p.id AS pedido_id, p.valor_total FROM clientes c LEFT JOIN pedidos p ON c.id = p.id_cliente WHERE p.status = 'faturado' AND p.valor_total > 500;

Query otimizada:
SELECT c.id, c.nome. p.id AS pedido_id, p.valor_total FROM cliente c JOIN pedidos p ON c.id = p.id_cliente WHERE p.status = 'faturado' AND p.valor_total > 500;
